// ── Setup: where the names below come from ──
import Anthropic from '@anthropic-ai/sdk';
import { readFileSync } from 'node:fs';
const client = new Anthropic();                           // one per app; reads ANTHROPIC_API_KEY
const POLICY = readFileSync('support-policy.md', 'utf8'); // 3,000 tokens, identical on every call
// Task: your own type with id, kind, text and someoneWaiting
// summarise(text): e.g. a cheap Haiku call that shortens a long input
// ── The example ──
function buildRequest(task: Task): Anthropic.MessageCreateParamsNonStreaming {
  const simple = task.kind === 'classify' || task.kind === 'faq';
  return {
    model: simple ? 'claude-haiku-5-5' : 'claude-sonnet-5-5',          // ① smallest model that passes (5.1)
    max_tokens: 1024,
    output_config: { effort: simple ? 'low' : 'high' },                 // ② effort to match the task (5.2)
    system: [{ type: 'text', text: POLICY,
               cache_control: { type: 'ephemeral' } }],                  // ③ cache the stable prefix (5.3)
    messages: [{ role: 'user', content: task.text }],
  };
}

async function run(tasks: Task[]) {
  for (let task of tasks.filter((t) => t.someoneWaiting)) {
    let req = buildRequest(task);
    const { input_tokens } = await client.messages.countTokens({
      model: req.model, system: req.system, messages: req.messages });
    if (input_tokens > 50_000) {                                        // ⑤ count before you pay (5.5)
      task = { ...task, text: await summarise(task.text) };
      req = buildRequest(task);
    }
    const reply = await client.messages.create(req);
    console.log(reply.usage.cache_read_input_tokens);                   // ③ > 0 means the cache hit
  }

  await client.messages.batches.create({                                // ④ batch what can wait: 50% off (5.4)
    requests: tasks.filter((t) => !t.someoneWaiting)
      .map((t) => ({ custom_id: t.id, params: buildRequest(t) })),
  });
}
